```csharp
using System.Globalization;
using System.Text.RegularExpressions;

namespace Notlar.Core;

public static partial class TagParser
{
    private static readonly CultureInfo Tr = CultureInfo.GetCultureInfo("tr-TR");

    [GeneratedRegex(@"(?<=^|\s)#([\p{L}\p{N}_-]+)")]
    private static partial Regex TagRegex();

    [GeneratedRegex(@"\s{2,}")]
    private static partial Regex MultiSpace();

    public static IReadOnlyList<string> Parse(string text) =>
        TagRegex().Matches(text).Select(m => Normalize(m.Groups[1].Value)).Distinct().ToList();

    public static string Strip(string text) =>
        MultiSpace().Replace(TagRegex().Replace(text, ""), " ").Trim();

    public static string Normalize(string tag) => tag.Trim().TrimStart('#').ToLower(Tr);
}
```

`src/Notlar.Core/ChangeSignal.cs`:
```csharp
namespace Notlar.Core;

public interface IChangeSignal
{
    void Notify();
}

public sealed class NullChangeSignal : IChangeSignal
{
    public static readonly NullChangeSignal Instance = new();
    public void Notify() { }
}

/// <summary>Süreçler arası "veri değişti" sinyali (named auto-reset event).</summary>
public sealed class NamedEventChangeSignal(string name) : IChangeSignal, IDisposable
{
    private readonly EventWaitHandle _handle = new(false, EventResetMode.AutoReset, name);

    public void Notify() => _handle.Set();
    public void Dispose() => _handle.Dispose();
}

/// <summary>Sinyali arka plan thread'inde bekler; her sinyalde onChange çağrılır (UI thread'e dispatch çağıranın işi).</summary>
public sealed class ChangeListener : IDisposable
{
    private readonly EventWaitHandle _handle;
    private readonly CancellationTokenSource _cts = new();
    private readonly Thread _thread;

    public ChangeListener(string name, Action onChange)
    {
        _handle = new EventWaitHandle(false, EventResetMode.AutoReset, name);
        _thread = new Thread(() => Loop(onChange)) { IsBackground = true, Name = "Notlar.ChangeListener" };
        _thread.Start();
    }

    private void Loop(Action onChange)
    {
        WaitHandle[] handles = [_handle, _cts.Token.WaitHandle];
        while (WaitHandle.WaitAny(handles) == 0)
            onChange();
    }

    public void Dispose()
    {
        _cts.Cancel();
        _thread.Join(TimeSpan.FromSeconds(1));
        _handle.Dispose();
        _cts.Dispose();
    }
}
```

```csharp
    private const string TimeFormat = "yyyy-MM-dd'T'HH:mm:ss.fffzzz";
    private const string Columns = "id, text, created_at, updated_at, done";
    private static readonly CultureInfo Inv = CultureInfo.InvariantCulture;

    private readonly string _connectionString;
    private readonly TimeProvider _time;
    private readonly IChangeSignal _signal;

    public NoteStore(string dbPath, TimeProvider time, IChangeSignal signal)
    {
        Directory.CreateDirectory(Path.GetDirectoryName(Path.GetFullPath(dbPath))!);
        // DefaultTimeout: SQLITE_BUSY durumunda 5 sn yeniden dener.
        _connectionString = new SqliteConnectionStringBuilder { DataSource = dbPath, DefaultTimeout = 5 }.ToString();
        _time = time;
        _signal = signal;

        using var conn = Open();
        using var cmd = conn.CreateCommand();
        cmd.CommandText = """
            PRAGMA journal_mode=WAL;
            CREATE TABLE IF NOT EXISTS notes (
                id         INTEGER PRIMARY KEY AUTOINCREMENT,
                text       TEXT    NOT NULL,
                created_at TEXT    NOT NULL,
                updated_at TEXT    NULL,
                done       INTEGER NOT NULL DEFAULT 0,
                deleted_at TEXT    NULL
            );
            CREATE INDEX IF NOT EXISTS ix_notes_created_at ON notes(created_at);
            """;
        cmd.ExecuteNonQuery();
    }

    public Note Add(string text)
    {
        var clean = Validate(text);
        long id;
        using (var conn = Open())
        using (var cmd = conn.CreateCommand())
        {
            cmd.CommandText = "INSERT INTO notes(text, created_at) VALUES($text, $now) RETURNING id";
            cmd.Parameters.AddWithValue("$text", clean);
            cmd.Parameters.AddWithValue("$now", Now());
            id = (long)cmd.ExecuteScalar()!;
        }
        _signal.Notify();
        return Get(id)!;
    }

    public Note? Get(long id)
    {
        using var conn = Open();
        using var cmd = conn.CreateCommand();
        cmd.CommandText = $"SELECT {Columns} FROM notes WHERE id = $id AND deleted_at IS NULL";
        cmd.Parameters.AddWithValue("$id", id);
        using var reader = cmd.ExecuteReader();
        return reader.Read() ? Map(reader) : null;
    }

    public IReadOnlyList<Note> List(NoteFilter filter)
    {
        using var conn = Open();
        using var cmd = conn.CreateCommand();
        var sql = new StringBuilder($"SELECT {Columns} FROM notes WHERE deleted_at IS NULL");
        if (filter.From is { } from)
        {
            sql.Append(" AND substr(created_at, 1, 10) >= $from");
            cmd.Parameters.AddWithValue("$from", from.ToString("yyyy-MM-dd", Inv));
        }
        if (filter.To is { } to)
        {
            sql.Append(" AND substr(created_at, 1, 10) <= $to");
            cmd.Parameters.AddWithValue("$to", to.ToString("yyyy-MM-dd", Inv));
        }
        if (filter.OnlyOpen)
            sql.Append(" AND done = 0");
        sql.Append(" ORDER BY id");
        cmd.CommandText = sql.ToString();

        var notes = new List<Note>();
        using (var reader = cmd.ExecuteReader())
            while (reader.Read())
                notes.Add(Map(reader));

        if (string.IsNullOrWhiteSpace(filter.Tag))
            return notes;
        var tag = TagParser.Normalize(filter.Tag);
        return notes.Where(n => n.Tags.Contains(tag)).ToList();
    }

    public Note Update(long id, string text)
    {
        Execute(id, "text = $value, updated_at = $now", Validate(text), deleted: false);
        return Get(id)!;
    }

    public Note SetDone(long id, bool done)
    {
        Execute(id, "done = $value", done ? 1 : 0, deleted: false);
        return Get(id)!;
    }

    public void Delete(long id) => Execute(id, "deleted_at = $now", null, deleted: false);

    public Note Restore(long id)
    {
        Execute(id, "deleted_at = NULL", null, deleted: true);
        return Get(id)!;
    }

    public int PurgeDeleted(TimeSpan olderThan)
    {
        using var conn = Open();
        using var cmd = conn.CreateCommand();
        cmd.CommandText = "DELETE FROM notes WHERE deleted_at IS NOT NULL AND deleted_at < $cutoff";
        cmd.Parameters.AddWithValue("$cutoff", Format(_time.GetLocalNow() - olderThan));
        return cmd.ExecuteNonQuery();
    }

    public string? Backup(string dir, int keep)
    {
        Directory.CreateDirectory(dir);
        var today = DateOnly.FromDateTime(_time.GetLocalNow().DateTime);
        var path = Path.Combine(dir, $"notlar-{today.ToString("yyyy-MM-dd", Inv)}.db");
        string? created = null;
        if (!File.Exists(path))
        {
            using var source = Open();
            using var target = new SqliteConnection(
                new SqliteConnectionStringBuilder { DataSource = path, Pooling = false }.ToString());
            target.Open();
            source.BackupDatabase(target);
            created = path;
        }
        foreach (var old in Directory.GetFiles(dir, "notlar-*.db").Order(StringComparer.Ordinal).Reverse().Skip(keep))
            File.Delete(old);
        return created;
    }

    private void Execute(long id, string set, object? value, bool deleted)
    {
        using (var conn = Open())
        using (var cmd = conn.CreateCommand())
        {
            cmd.CommandText = $"UPDATE notes SET {set} WHERE id = $id AND deleted_at IS {(deleted ? "NOT NULL" : "NULL")}";
            cmd.Parameters.AddWithValue("$id", id);
            cmd.Parameters.AddWithValue("$now", Now());
            if (value is not null)
                cmd.Parameters.AddWithValue("$value", value);
            if (cmd.ExecuteNonQuery() == 0)
                throw new NoteNotFoundException(id);
        }
        _signal.Notify();
    }

    private SqliteConnection Open()
    {
        var conn = new SqliteConnection(_connectionString);
        conn.Open();
        return conn;
    }

    private static string Validate(string? text)
    {
        var clean = text?.Trim();
        if (string.IsNullOrEmpty(clean))
            throw new ArgumentException("Not metni boş olamaz.");
        return clean;
    }

    private string Now() => Format(_time.GetLocalNow());

    private static string Format(DateTimeOffset value) => value.ToString(TimeFormat, Inv);

    private static DateTimeOffset Parse(string value) => DateTimeOffset.ParseExact(value, TimeFormat, Inv);

    private static Note Map(SqliteDataReader r) => new(
        r.GetInt64(0),
        r.GetString(1),
        Parse(r.GetString(2)),
        r.IsDBNull(3) ? null : Parse(r.GetString(3)),
        r.GetInt64(4) != 0);
}
```

```csharp
using System.Globalization;

namespace Notlar.Core;

public sealed record DayGroup(DateOnly Date, IReadOnlyList<Note> Notes);

public static class DayGrouper
{
    private static readonly CultureInfo Tr = CultureInfo.GetCultureInfo("tr-TR");

    public static IReadOnlyList<DayGroup> Group(IEnumerable<Note> notes) =>
        notes.GroupBy(n => n.Day)
             .OrderByDescending(g => g.Key)
             .Select(g => new DayGroup(g.Key, g.OrderBy(n => n.CreatedAt).ThenBy(n => n.Id).ToList()))
             .ToList();

    public static string FormatDate(DateOnly date) => date.ToString("d MMMM yyyy, dddd", Tr);

    public static string Header(DateOnly date, DateOnly today)
    {
        if (date == today) return $"BUGÜN · {FormatDate(date)}";
        if (date == today.AddDays(-1)) return $"DÜN · {FormatDate(date)}";
        return FormatDate(date);
    }
}
```

```csharp
            return null;
        if (DateOnly.TryParseExact(value.Trim(), "yyyy-MM-dd", CultureInfo.InvariantCulture, DateTimeStyles.None, out var date))
            return date;
        throw new ArgumentException($"Geçersiz tarih '{value}', yyyy-MM-dd bekleniyor.");
    }

    private static string Run(Func<string> action)
    {
        try
        {
            return action();
        }
        catch (Exception ex) when (ex is NoteNotFoundException or ArgumentException)
        {
            throw new McpException(ex.Message);
        }
    }
}
```

```csharp
            Directory.CreateDirectory(AppPaths.DataDir);
            SetEnabled(true);
            File.WriteAllText(MarkerPath, DateTime.Now.ToString("o"));
        }
        else if (IsEnabled())
        {
            SetEnabled(true);
        }
    }
```

```csharp
using System.Collections.ObjectModel;
using System.Diagnostics;
using System.Globalization;
using System.IO;

using System.Windows;
using System.Windows.Controls;

using System.Windows.Threading;
using Microsoft.Data.Sqlite;
using Notlar.Core;

namespace Notlar.App;

public partial class PanelWindow : Window
{
    private static readonly StringComparer TrComparer = StringComparer.Create(CultureInfo.GetCultureInfo("tr-TR"), ignoreCase: true);

    private readonly NoteStore _store;
    private readonly DispatcherTimer _undoTimer = new() { Interval = TimeSpan.FromSeconds(5) };
    private readonly DispatcherTimer _errorTimer = new() { Interval = TimeSpan.FromSeconds(4) };
    private string? _tagFilter;
    private long? _undoId;
    private bool _reloadPending;
    private DateTime _lastHiddenAt = DateTime.MinValue;

    public ObservableCollection<DayGroupItem> Groups { get; } = [];
    public ObservableCollection<TagChip> Tags { get; } = [];

    public PanelWindow(NoteStore store)
    {
        _store = store;
        InitializeComponent();
        DataContext = this;
        _undoTimer.Tick += (_, _) => HideUndo();
        _errorTimer.Tick += (_, _) => { _errorTimer.Stop(); ErrorBar.Visibility = Visibility.Collapsed; };
    }

    private bool AnyEditing => Groups.SelectMany(g => g.Notes).Any(n => n.IsEditing);

    /// <summary>Tray sol tık. Panel açıkken tıklama önce Deactivated'ı tetikler; hemen yeniden açmayı engelle.</summary>
    public void Toggle()
    {
        if (IsVisible) HidePanel();
        else if ((DateTime.Now - _lastHiddenAt).TotalMilliseconds > 300) ShowNearTray();
    }

    public void ShowNearTray()
    {
        var area = SystemParameters.WorkArea;
        Left = area.Right - Width - 8;
        Top = area.Bottom - Height - 8;
        Reload();
        Show();
        Activate();
        InputBox.Focus();
    }

    private void HidePanel()
    {
        foreach (var note in Groups.SelectMany(g => g.Notes))
            note.IsEditing = false;
        Hide();
        _lastHiddenAt = DateTime.Now;
    }

    public void Reload()
    {
        if (AnyEditing)
        {
            _reloadPending = true;
            return;
        }
        _reloadPending = false;
        Try(() =>
        {
            var all = _store.List(new NoteFilter());
            var tags = all.SelectMany(n => n.Tags).Distinct().Order(TrComparer).ToList();
            if (_tagFilter is not null && !tags.Contains(_tagFilter))
                _tagFilter = null;

            Tags.Clear();
            Tags.Add(new TagChip(null, "Tümü", _tagFilter is null));
            foreach (var tag in tags)
                Tags.Add(new TagChip(tag, "#" + tag, tag == _tagFilter));
            TagRow.Visibility = tags.Count > 0 ? Visibility.Visible : Visibility.Collapsed;

            var visible = _tagFilter is null ? all : all.Where(n => n.Tags.Contains(_tagFilter)).ToList();
            var today = DateOnly.FromDateTime(DateTime.Now);
            Groups.Clear();
            foreach (var group in DayGrouper.Group(visible))
                Groups.Add(new DayGroupItem(DayGrouper.Header(group.Date, today), group.Notes.Select(n => new NoteItem(n)).ToList()));
            EmptyText.Visibility = Groups.Count == 0 ? Visibility.Visible : Visibility.Collapsed;
        });
    }

    public void ExportMarkdown() => Try(() =>
    {
        var documents = Environment.GetFolderPath(Environment.SpecialFolder.MyDocuments);
        var path = Path.Combine(documents, $"Notlar-{DateTime.Now.ToString("yyyy-MM-dd", CultureInfo.InvariantCulture)}.md");
        File.WriteAllText(path, MarkdownExporter.Export(_store.List(new NoteFilter())), new UTF8Encoding(false));
        Process.Start("explorer.exe", $"/select,\"{path}\"");
    });

    private void Window_Deactivated(object? sender, EventArgs e) => HidePanel();

    private void Window_PreviewKeyDown(object sender, KeyEventArgs e)
    {
        if (AnyEditing) return; // Düzenleme kutusu Enter/Esc'i kendisi işler.
        if (e.Key == Key.Escape)
        {
            HidePanel();
            e.Handled = true;
        }
        else if (e.Key == Key.Delete && Keyboard.FocusedElement is FrameworkElement { DataContext: NoteItem item })
        {
            DeleteNote(item);
            e.Handled = true;
        }
    }

    private void InputBox_KeyDown(object sender, KeyEventArgs e)
    {
        if (e.Key != Key.Enter) return;
        e.Handled = true;
        if (string.IsNullOrWhiteSpace(InputBox.Text)) return;
        Try(() =>
        {
            _store.Add(InputBox.Text);
            InputBox.Clear();
            Reload();
        });
    }

    private void TagChip_Click(object sender, RoutedEventArgs e)
    {
        if (sender is FrameworkElement { DataContext: TagChip chip })
        {
            _tagFilter = chip.Tag;
            Reload();
        }
    }

    private void Row_MouseLeftButtonDown(object sender, MouseButtonEventArgs e) => ((UIElement)sender).Focus();

    private void Done_Click(object sender, RoutedEventArgs e)
    {
        if (sender is CheckBox { DataContext: NoteItem item } box)
            Try(() => { _store.SetDone(item.Id, box.IsChecked == true); Reload(); });
    }

    private void NoteText_MouseLeftButtonDown(object sender, MouseButtonEventArgs e)
    {
        if (e.ClickCount != 2 || sender is not FrameworkElement { DataContext: NoteItem item }) return;
        item.EditText = item.Note.Text;
        item.IsEditing = true;
        e.Handled = true;
    }

    private void EditBox_IsVisibleChanged(object sender, DependencyPropertyChangedEventArgs e)
    {
        if (sender is TextBox { IsVisible: true } box)
            Dispatcher.InvokeAsync(() => { box.Focus(); box.SelectAll(); }, DispatcherPriority.Input);
    }

    private void EditBox_KeyDown(object sender, KeyEventArgs e)
    {
        if (sender is not TextBox { DataContext: NoteItem item }) return;
        if (e.Key == Key.Enter)
        {
            e.Handled = true;
            var text = item.EditText.Trim();
            item.IsEditing = false;
            if (text.Length > 0 && text != item.Note.Text)
                Try(() => _store.Update(item.Id, text));
            Reload();
        }
        else if (e.Key == Key.Escape)
        {
            e.Handled = true;
            CancelEdit(item);
        }
    }

    private void EditBox_LostKeyboardFocus(object sender, KeyboardFocusChangedEventArgs e)
    {
        if (sender is TextBox { DataContext: NoteItem { IsEditing: true } item })
            CancelEdit(item);
    }

    private void CancelEdit(NoteItem item)
    {
        item.IsEditing = false;
        if (_reloadPending) Reload();
    }

    private void DeleteButton_Click(object sender, RoutedEventArgs e)
    {
        if (sender is FrameworkElement { DataContext: NoteItem item })
            DeleteNote(item);
    }

    private void DeleteNote(NoteItem item) => Try(() =>
    {
        _store.Delete(item.Id);
        _undoId = item.Id;
        UndoBar.Visibility = Visibility.Visible;
        _undoTimer.Stop();
        _undoTimer.Start();
        Reload();
    });

    private void Undo_Click(object sender, RoutedEventArgs e)
    {
        if (_undoId is { } id)
            Try(() => { _store.Restore(id); Reload(); });
        HideUndo();
    }

    private void HideUndo()
    {
        _undoTimer.Stop();
        _undoId = null;
        UndoBar.Visibility = Visibility.Collapsed;
    }

    private void Try(Action action)
    {
        try
        {
            action();
        }
        catch (Exception ex)
        {
            AppLog.Write(ex);
            ErrorText.Text = ex switch
            {
                NoteNotFoundException or ArgumentException => ex.Message,
                SqliteException => $"Veritabanı hatası. Yedekler: {AppPaths.BackupDir}",
                _ => $"Bir hata oluştu, ayrıntı: {AppPaths.LogPath}",
            };
            ErrorBar.Visibility = Visibility.Visible;
            _errorTimer.Stop();
            _errorTimer.Start();
        }
    }
```

```csharp
using System;
using System.Threading;
using System.Windows;
using Microsoft.Win32;
using Notlar.Core;

namespace Notlar.App;

public partial class App : Application
{
    private const string MutexName = @"Local\Notlar.App.Tek";
    private const string ShowEventName = @"Local\Notlar.App.Goster";

    private Mutex? _mutex;
    private EventWaitHandle? _showEvent;
    private RegisteredWaitHandle? _showWait;
    private ChangeListener? _listener;
    private TrayIcon? _tray;
    private PanelWindow? _panel;

    protected override void OnStartup(StartupEventArgs e)
    {
        base.OnStartup(e);

        _mutex = new Mutex(initiallyOwned: true, MutexName, out var isFirstInstance);
        _showEvent = new EventWaitHandle(false, EventResetMode.AutoReset, ShowEventName);
        if (!isFirstInstance)
        {
            // Zaten çalışıyor: mevcut örneğe paneli açmasını söyle ve çık.
            _showEvent.Set();
            Shutdown();
            return;
        }

        DispatcherUnhandledException += (_, args) => { AppLog.Write(args.Exception); args.Handled = true; };
        AppDomain.CurrentDomain.UnhandledException += (_, args) => AppLog.Write(args.ExceptionObject as Exception);

        Theme.Apply(Resources);
        SystemEvents.UserPreferenceChanged += OnUserPreferenceChanged;

        NoteStore store;
        try
        {
            store = new NoteStore(AppPaths.DbPath, TimeProvider.System, new NamedEventChangeSignal(AppPaths.ChangeEventName));
        }
        catch (Exception ex)
        {
            AppLog.Write(ex);
            MessageBox.Show($"Not veritabanı açılamadı.\n\n{ex.Message}\n\nYedekler: {AppPaths.BackupDir}", "Notlar",
                MessageBoxButton.OK, MessageBoxImage.Error);
            Shutdown();
            return;
        }

        try
        {
            store.PurgeDeleted(TimeSpan.FromDays(7));
            store.Backup(AppPaths.BackupDir, keep: 7);
            AutoStart.EnsureRegistration();
        }
        catch (Exception ex)
        {
            AppLog.Write(ex);
        }

        _panel = new PanelWindow(store);
        _tray = new TrayIcon(
            onToggle: _panel.Toggle,
            onOpen: _panel.ShowNearTray,
            onExport: _panel.ExportMarkdown,
            onExit: ExitApp);
        _listener = new ChangeListener(AppPaths.ChangeEventName, () => Dispatcher.InvokeAsync(_panel.Reload));
        _showWait = ThreadPool.RegisterWaitForSingleObject(
            _showEvent, (_, _) => Dispatcher.InvokeAsync(_panel.ShowNearTray), null, Timeout.Infinite, executeOnlyOnce: false);
    }

    private void OnUserPreferenceChanged(object sender, UserPreferenceChangedEventArgs e)
    {
        if (e.Category == UserPreferenceCategory.General)
            Dispatcher.InvokeAsync(() => Theme.Apply(Resources));
    }

    private void ExitApp()
    {
        SystemEvents.UserPreferenceChanged -= OnUserPreferenceChanged;
        _showWait?.Unregister(null);
        _listener?.Dispose();
        _tray?.Dispose();
        _mutex?.ReleaseMutex();
        Shutdown();
    }
}
```

```powershell
#Requires -Version 5.1
# Notlar'ı derler, %LOCALAPPDATA%\Programs\Notlar'a kurar, Claude Code'a MCP olarak kaydeder ve başlatır.
$ErrorActionPreference = 'Stop'
$root = $PSScriptRoot
$target = Join-Path $env:LOCALAPPDATA 'Programs\Notlar'
$appDir = Join-Path $target 'app'
$mcpDir = Join-Path $target 'mcp'
$mcpExe = Join-Path $mcpDir 'NotlarMcp.exe'

Write-Host '> Çalışan Notlar süreçleri kapatılıyor...'
Get-Process NotlarApp, NotlarMcp -ErrorAction SilentlyContinue | Stop-Process -Force
Start-Sleep -Milliseconds 500

Write-Host '> Derleniyor...'
dotnet publish (Join-Path $root 'src\Notlar.App\Notlar.App.csproj') -c Release -o $appDir --nologo
if ($LASTEXITCODE -ne 0) { throw 'NotlarApp derlenemedi.' }
dotnet publish (Join-Path $root 'src\Notlar.Mcp\Notlar.Mcp.csproj') -c Release -o $mcpDir --nologo
if ($LASTEXITCODE -ne 0) { throw 'NotlarMcp derlenemedi.' }

Write-Host '> Claude Code MCP kaydı...'
cmd /c "claude mcp remove notlar --scope user >nul 2>&1"
claude mcp add notlar --scope user -- $mcpExe
if ($LASTEXITCODE -ne 0) { throw 'MCP kaydı başarısız.' }

Write-Host '> CLAUDE.md yönergesi...'
$claudeMd = Join-Path $env:USERPROFILE '.claude\CLAUDE.md'
$line = '- Kullanıcı "not al", "not et", "kaydet" dediğinde notu `notlar` MCP sunucusunun `not_ekle` aracıyla ekle.'
if (-not (Test-Path $claudeMd) -or -not (Select-String -Path $claudeMd -SimpleMatch 'not_ekle' -Quiet)) {
    $utf8 = New-Object System.Text.UTF8Encoding $false
    [System.IO.File]::AppendAllText($claudeMd, [Environment]::NewLine + $line + [Environment]::NewLine, $utf8)
}

Write-Host '> Başlatılıyor...'
Start-Process (Join-Path $appDir 'NotlarApp.exe')
Write-Host 'Tamam. Tray ikonuna tıkla; Claude Code oturumunu yeniden başlatınca "not al: ..." diyebilirsin.'
```
